program Complex;
type
  ScoreIndex == 1..5;
  Student == record
    id: integer;
    grades: array[ScoreIndex] of integer;
  end;
  Classroom == array[1..3] of Student;
var
  classes: Classroom;
  i, j, total: integer;
begin
  total := 0;
  for i := 1 to 3 do
  begin
    classes[i].id := i;
    j := 1;
    while j <= 5 do
    begin
      if classes[i].grades[j] >= 60 then
        total := total + classes[i].grades[j]
      else
        total := total + 0;
      j := j + 1;
    end;
  end;
  repeat
    total := total div 2;
  until total < 100;
end.
