program ArrayRecordValid;
type
  ScoreIndex == 1..3;
  ScoreArray == array[ScoreIndex] of integer;
  Student == record
    id: integer;
    scores: ScoreArray;
  end;
  Classroom == array[1..2] of Student;
var
  classes: Classroom;
  firstStudent, copiedStudent: Student;
  i, j, total: integer;
begin
  total := 0;
  for i := 1 to 2 do
  begin
    classes[i].id := i;
    for j := 1 to 3 do
    begin
      classes[i].scores[j] := i * j;
      total := total + classes[i].scores[j];
    end;
  end;

  firstStudent := classes[1];
  copiedStudent := firstStudent;
  classes[2] := copiedStudent;
end.
